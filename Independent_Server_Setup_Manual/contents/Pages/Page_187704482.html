require 'json'
require 'openssl'
require 'base64'
require 'jwt'
 
ISSUER='https://e97b8a9d672e4ce4845ec6947cd66ef6-sb.baas.nintendo.com'
JWKS_URI='https://e97b8a9d672e4ce4845ec6947cd66ef6-sb.baas.nintendo.com/1.0.0/certificates'
APPLICATION_ID='0123456789012345'
ALGORITHM='RS256'
REQUIRED_VERSION=0 # Option
 
 
def get_jwks(jwks_uri)
  `curl -k -X GET #{jwks_uri} 2> /dev/null`
end
 
 
def public_key(jwks_uri, kid)
  jwks = get_jwks(jwks_uri)
  keys = JSON.parse( jwks )['keys']
  jwk  = keys.find{ |k| k['kid'] == kid } || "{}"
  OpenSSL::X509::Certificate.new(Base64.strict_decode64(jwk['x5c'][0])).public_key
end
 
 
def main
  id_token = ARGV[0]
  # 1. It is JWT format.
   _, unverified_header = JWT.decode(id_token, nil, false, { verify_expiration: false })
 
  # 2. The alg specified in the JOSE Header is RS256.
  raise "Invalid alg: #{unverified_header['alg']}" unless unverified_header['alg'] == ALGORITHM
 
  # 3-1. The jku URI matches the endpoint URI of the key public endpoint that was registered in advance.
  raise "Invalid jku: #{unverified_header['jku']}" unless JWKS_URI === unverified_header['jku']
 
  # 3-2. JWS signature verification is successful with the key or certificate corresponding to the kid specified in the JWS Header.
  key = public_key(unverified_header['jku'], unverified_header['kid'])
  payload, header = JWT.decode(id_token, key, true, { algorithm: ALGORITHM,  verify_expiration: false })
 
  # 4. The iss matches the Issuer registered in advance.
  raise "Invalid iss: #{payload['iss']}" unless payload['iss'] == ISSUER
 
  # 5. The iat is before the current time.
  raise 'iat is not the past time' unless Time.now.to_i >= payload['iat']
 
  # 6. The exp is after the current time.
  raise 'This token is already expired' unless Time.now.to_i < payload['exp']
 
  # 7. The nintendo.ai value and the Application ID that accepts connection match.
  raise "Invalid Application : #{payload['nintendo']['ai']}" unless payload['nintendo']['ai'] == APPLICATION_ID
 
  # Option. Equal to or greater than the application version for which the nintendo.av value permits connections (option).
  # Independent judgment is required if you want to restrict connections from earlier versions of the application immediately after a patch is released.
  raise "Invalid Version: #{payload['nintendo']['av']}" unless payload['nintendo']['av'].to_i >= REQUIRED_VERSION

  puts "=== ID Token ==="
  puts "token   : #{id_token}"
  puts "jku     : #{header['jku']}"
  puts "kid     : #{header['kid']}"
  puts "=== Decoded Token ==="
  puts "header  : #{header}"
  puts "payload : #{payload}"
end
main

#!/usr/bin/env perl
use strict;
use warnings;

use Carp;
use JSON::XS qw/decode_json/;
use JSON::WebToken qw/decode_jwt/;

use Crypt::OpenSSL::RSA;
use Crypt::OpenSSL::Bignum;
use MIME::Base64 qw/decode_base64url/;

our $ISSUER           = 'https://e97b8a9d672e4ce4845ec6947cd66ef6-sb.baas.nintendo.com';
our $JWKS_URI         = 'https://e97b8a9d672e4ce4845ec6947cd66ef6-sb.baas.nintendo.com/1.0.0/certificates';
our $APPLICATION_ID = '0123456789012345';
our $ALGORITHM      = 'RS256';
our $REQUIRED_VERSION = 0; # Option

sub get_jwks {
    my $jwks_uri = shift;
    return `curl -k -X GET $jwks_uri 2> /dev/null`;
}

sub public_key {
    my ($jwks_uri, $kid) = @_;

    my $jwks = get_jwks($jwks_uri);
    my $keys = decode_json($jwks)->{keys};

    my ($jwk) = grep { $_->{kid} eq $kid } @$keys;

    my $rsa = Crypt::OpenSSL::RSA->new_key_from_parameters(
        Crypt::OpenSSL::Bignum->new_from_bin(decode_base64url($jwk->{n})),
        Crypt::OpenSSL::Bignum->new_from_bin(decode_base64url($jwk->{e})),
    );
    return $rsa->get_public_key_x509_string;
}

main($ARGV[0]);

sub main {
    my $id_token = shift;

    my $jwt_claims;
    eval {
        $jwt_claims = decode_jwt (
            $id_token, sub {
                my ($header, $claims) = @_;

                # The alg specified in the JOSE Header must be RS256.
                if ($header->{alg} ne $ALGORITHM) {
                    croak sprintf("Unsupported alg (%s)", $header->{alg} || "");
                }

                # The jku URI matches the endpoint URI of the key public endpoint that was registered in advance.
                if ($header->{jku} !~ /$JWKS_URI/) {
                    croak sprintf("Not acceptable jku (jku: %s)", $header->{jku});
                }

                # The iss value matches the preregistered ISSUER.
                if ($claims->{iss} ne $ISSUER) {
                    croak sprintf("Not acceptable iss (iss: %s)", $claims->{iss});
                }

                my $current_time = time;

                # The value of iat is a value before the current time.
                if (exists $claims->{iat} && $claims->{iat} >= $current_time) {
                    croak sprintf("Invalid iat (iat: %s)", $claims->{iat});
                }
                # The value of exp is after the current time.
                if (exists $claims->{exp} && $claims->{exp} < $current_time) {
                    croak sprintf("Expired JWT (exp: %s)", $claims->{exp});
                }
                # The value of nintendo.ai matches the Application ID to allow a connection.
                if ($claims->{nintendo}->{ai} ne $APPLICATION_ID) {
                    croak sprintf("Not acceptable application_id (nintendo.ai: %s)", $claims->{nintendo}->{ai});
                }
                # Option. Equal to or greater than the application version for which the nintendo.av value permits connections (option).
                # Independent judgment is required if you want to restrict connections from earlier versions of the application immediately after a patch is released.
                if ($claims->{nintendo}->{av} < $REQUIRED_VERSION) {
                    croak sprintf("Not acceptable application_version (nintendo.av: %s)", $claims->{nintendo}->{av});
                }

                return public_key($header->{jku}, $header->{kid});
            }
        );
    };
    if (my $e = $@) {
        warn $e;
        return;
    }
    return $jwt_claims;
}
